# ShedOS Base System Packages

# Core System
base

# Supplies BTF info for DKMS builds against the kernel headers.
pahole

# Primary kernel — Arch linux-zen (desktop-tuned scheduler/HZ). The
# -headers package is an explicit root so DKMS modules can build against
# it — nvidia's on the boxes that keep the stack, and any user-added ones.
linux-zen
linux-zen-headers

# Fallback kernel — stock linux, so a worst-case linux-zen breakage still
# leaves a bootable entry. linux-headers covers its own DKMS builds.
linux
linux-headers

# Security — AppArmor mandatory access control, on by default. The LSM is
# switched on via the kernel lsm= cmdline (baked in by the installer, and
# backfilled onto existing installs by shedos-system); apparmor.service
# loads the /etc/apparmor.d profile set at boot. Pulls audit for logging.
apparmor

# Bootloader
limine
efibootmgr

# Secure Boot + measured boot. sbctl manages the db/KEK/PK key set and
# verifies signatures; tpm2-tools backs passphrase-free TPM2 unlock;
# systemd-ukify is the single signer that sbsigns and PCR-11-signs the UKI;
# sbsigntools provides the sbverify that gates each UKI onto the ESP;
# libfido2 backs FIDO2 security-key unlock and the sd-encrypt initramfs bundle.
sbctl
tpm2-tools
systemd-ukify
sbsigntools
libfido2
